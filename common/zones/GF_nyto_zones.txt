# ==================== Embers (Gloomy World - Pre-Spring) Zones ====================

zone_GF_nyto_embers_city = {
	icon = GFX_district_specialization_urban
	base_buildtime = 120
	potential = {
		uses_district_set = GF_nyto_embers_gloomy_world
		NOT = { has_planet_flag = GFnyto_Spring_wakes_up }
	}
	unlock = {
		always = yes
	}
	resources = {
		category = planet_zones
		cost = {
			minerals = @zone_cost
		}
	}
	max_buildings = 3
	zone_sets = {
		zone_GF_nyto_embers_city
	}
	include = {
		all
	}
	excluded_building_sets = {
		generator_automation
		mining_automation
		farming_automation
	}
	district_planet_modifier = {
		planet_housing_add = 3000
		planet_amenities_add = 500
	}
	planet_modifier = {
		zone_building_slots_add = 3
	}
}
