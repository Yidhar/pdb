GF_nyto_elid_attack = {
	if = {
		limit = {
			is_scope_type = planet
			exists = owner
			exists = event_target:EILD_country@root
			exists = event_target:ELID_species@root
		}
		if = {
			limit = {
				has_modifier = GFnyto_flower_period
			}
			change_variable = {
				which = GFnyto_Eild_level
				value = 5
			}
		}
		if = {
			limit = {
				check_variable = {
					which = GFnyto_Eild_level
					value < 50
				}
			}
			while = {
				count = GFnyto_Eild_level
				random_list = {
					50 = {
						create_army = {
							name = "EILD"
							owner = event_target:EILD_country@root
							species = event_target:ELID_species@root
							type = "GF_nyto_eild_army_1"
						}
					}
					50 = {
						create_army = {
							name = "EILD"
							owner = event_target:EILD_country@root
							species = event_target:ELID_species@root
							type = "GF_nyto_eild_army_2"
						}
					}
					1 = {
						create_army = {
							name = "EILD"
							owner = event_target:EILD_country@root
							species = event_target:ELID_species@root
							type = "GF_nyto_eild_army_3"
						}
					}
				}
			}
		}
		else_if = {
			limit = {
				check_variable = {
					which = GFnyto_Eild_level
					value >= 50
				}
				check_variable = {
					which = GFnyto_Eild_level
					value < 100
				}
			}
			set_variable = {
				which = GFnyto_Eild_level_rar
				value = GFnyto_Eild_level
			}
			multiply_variable = {
				which = GFnyto_Eild_level_rar
				value = 0.5
			}
			event_target:EILD_country@root = {
				add_modifier = {
					modifier = GF_nyto_Elid_level
					multiplier = prev.GFnyto_Eild_level_rar
				}
			}
			while = {
				count = GFnyto_Eild_level_rar
				create_army = {
					name = "EILD"
					owner = event_target:EILD_country@root
					species = event_target:ELID_species@root
					type = "GF_nyto_eild_army_3"
				}
			}
		}
		else_if = {
			limit = {
				check_variable = {
					which = GFnyto_Eild_level
					value >= 100
				}
			}
			set_variable = {
				which = GFnyto_Eild_level_rar
				value = GFnyto_Eild_level
			}
			event_target:EILD_country@root = {
				add_modifier = {
					modifier = GF_nyto_Elid_level2
					multiplier = prev.GFnyto_Eild_level_rar
				}
			}
			while = {
				count = GFnyto_Eild_level_rar
				create_army = {
					name = "EILD"
					owner = event_target:EILD_country@root
					species = event_target:ELID_species@root
					type = "GF_nyto_eild_army_3"
				}
			}
		}
	}
}
