can_destroy_planet_with_tech_GFnyto_accelerating_coil3 = {
    optimize_memory
    custom_tooltip = {
        fail_text = is_not_a_standard_planet_or_megastructure
        is_a_planet_megastructure = yes
    }
    custom_tooltip = {
        fail_text = is_shielded
        NOR = {
            is_planet_class = pc_shielded
            is_planet_class = pc_ringworld_shielded
            is_planet_class = pc_habitat_shielded
        }
    }
}

is_nyto_country = {
    optimize_memory
    OR = {
        owner_main_species = {
            is_species_class = nyto
        }
        has_country_flag = nyto_empire
    }
}

is_mechanical_empire = {
    optimize_memory
    OR = {
        owner_species = { has_trait = trait_mechanical }
        has_country_flag = synthetic_empire
        is_nyto_country = yes
    }
}

is_individual_machine = {
    optimize_memory
    OR = {
        AND = {
            founder_species = {
                is_archetype = MACHINE
            }
            is_gestalt = no
        }
        is_nyto_country = yes
    }
}

is_spring = {
    optimize_memory
    text = is_spring_desc
    check_variable_arithmetic = {
        which = month
        add = value:GF_nyto_get_game_time_month
        value >= 3
    }
    check_variable_arithmetic = {
        which = month
        add = value:GF_nyto_get_game_time_month
        value <= 5
    }
}

is_winter = {
    optimize_memory
    text = is_winter_desc
    OR = {
        check_variable_arithmetic = {
            which = month
            add = value:GF_nyto_get_game_time_month
            value == 12
        }
        check_variable_arithmetic = {
            which = month
            add = value:GF_nyto_get_game_time_month
            value <= 2
        }
    }
}
